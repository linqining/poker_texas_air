# zchain-recursion.slice —— zchain monad 递归证明/链节点资源隔离 slice
# 建于 2026-09-29。目的：把递归证明与链节点负载关进硬顶 cgroup，保证生产 texas.service 不受影响。
# 预算推导（2026-09-29 实测）：available 5806M；texas.service MemoryCurrent≈1029M、monad 栈常驻≈27M。
#   slice 上限 5200M = monad-leaf@ / monad-fold@ 各自 MemoryMax=4600M（必须串行跑）+ 预留编排类 512M + 内核开销。
#   CPUQuota=300%：4 核留 1 核给 texas/nginx/系统；CPUWeight/IOWeight=60（texas 缺省 100 优先）。
#   MemorySwapMax=0：宁可 cgroup OOM-kill 重试，不可 swap 风暴拖垮同盘 texas IO（9-28 磁盘事故同因教训）。
# 回退：systemctl stop 成员单元 && rm /etc/systemd/system/zchain-recursion.slice && systemctl daemon-reload
[Unit]
Description=zchain-monad recursion-proof & node workload slice (hard-capped; protects texas.service)
Documentation=file:/opt/zchain-monad/docs/node-unit-migration.md

[Slice]
CPUAccounting=true
MemoryAccounting=true
IOAccounting=true
CPUQuota=300%
CPUWeight=60
MemoryMax=5200M
MemorySwapMax=0
IOWeight=60

# This is a drop-in unit file extension, created via "systemctl set-property"

#   systemd-run --scope -p MemoryMax=4600M -p MemorySwapMax=0 -p CPUQuota=300% -p CPUWeight=60 \
#     -p LogRateLimitIntervalSec=30 ./leaf-prover <job>   # RAYON_NUM_THREADS ∈ {2,3} 各跑一次
#   判定：内存不足时 cgroup OOM 只杀探针进程（不波及 texas）；全程 texas.service active
#   且 MemoryCurrent 漂移 <10% 才算过。探针不过 → 叶层永久留在 36GB 工位（不阻塞上线）。
